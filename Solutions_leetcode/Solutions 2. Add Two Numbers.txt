/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode dummy(0);       //вспомогательный начальный узел
        ListNode* tail = &dummy; //указатель на последний узел 
        int tr = 0;

        // Пока есть цифры в любом списке или есть перенос
        while (l1 != nullptr || l2 != nullptr || tr != 0) {
            int a = (l1 != nullptr) ? l1->val : 0;  //извлечение цифры из узла из l1 
            int b = (l2 != nullptr) ? l2->val : 0;  //извлечение цифры из узла из l2
            int sum = a + b + tr;

            // новая цифра результата
            tail->next = new ListNode(sum % 10);
            tail = tail->next;
            tr = sum / 10;

            if (l1) l1 = l1->next;
            if (l2) l2 = l2->next;
        }
        return dummy.next;
    }    
};
