class Solution {
public:
    int reverse(int x) {
        int N = 0;
    while (x != 0) {
        int p = x % 10;   // извлечение последней цифры
        x /= 10;          // удаление последней цифруы из х

        // Проверка переполнения до выполнения N = N * 10 + p
        if (N > INT_MAX / 10 || (N == INT_MAX / 10 && p > 7)) return 0;
        if (N < INT_MIN / 10 || (N == INT_MIN / 10 && p < -8)) return 0;
        N = N * 10 + p;
    }
    return N;
    }

int main() {
    int x;
    std::cin >> x;
    std::cout << reverse(x) << std::endl;
    return 0;    
    }
};
