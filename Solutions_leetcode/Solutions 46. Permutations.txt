class Solution {
public:
    vector<vector<int>> permute(vector<int>& nums) {
        vector<vector<int>> N;
        sort(nums.begin(), nums.end()); // для начала с первой перестановки
        do {
            N.push_back(nums);
        } while (next_permutation(nums.begin(), nums.end()));
        return N;
    }
};
